title DublinBikes - Frontend React Components Diagram

package "React Frontend Subsystem (Pages)" {
  class Login <<FC>> {
    -identifier : String [State]
    -password : String [State]
    -loading : Boolean [State]
    +handleSubmit(e : Event) : Promise<void>
  }

  class Register <<FC>> {
    -username : String [State]
    -email : String [State]
    -password : String [State]
    -loading : Boolean [State]
    +handleSubmit(e : Event) : Promise<void>
  }

  class VerifyEmail <<FC>> {
    -verificationCode : String [State]
    -loading : Boolean [State]
    +handleSubmit(e : Event) : Promise<void>
  }

  class Profile <<FC>> {
    -userProfile : Map [State]
    -loading : Boolean [State]
    +handleLogout() : void
  }

  class Maps <<FC>> {
    -mapInstance : Object [State]
    -stations : List [State]
    -selectedStation : Map [State]
    -journeyResult : Map [State]
    -stationPredictions : List [State]
    +handleMapLoad(map : Object) : void
    +handleStationClick(station : Map) : void
    +handlePlanJourney(start : String, end : String) : Promise<void>
  }

  class Chat <<FC>> {
    -messages : List<Map> [State]
    -inputValue : String [State]
    -isLoading : Boolean [State]
    +handleSendMessage(e : Event) : Promise<void>
  }

  class Home <<FC>> {
    -weatherData : Map [State]
    -systemStats : Map [State]
  }
}
