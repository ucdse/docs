@startuml
title DublinBikes - AI Chatbot Interaction Flow

actor User
participant "chatPage : ChatPage" as UI
participant "chatRoutes : ChatRoutes" as API
participant "chatService : ChatService" as CS
participant "sessionRecord : Session" as SessionObj
participant "historyRecord : ChatHistory" as HistoryObj
participant "llmClient : LangChainLLM" as LLM

User -> UI: submitMessage(user_message)
activate UI

UI -> API: chat(session_id, user_message)
activate API

API -> API: _require_auth()

API -> CS: generate_chat_response(session_id, user_message, user_id)
activate CS

CS -> CS: _ensure_session(session_id, user_id)
activate CS
CS -> SessionObj: find_or_create(session_id, user_id)
activate SessionObj
SessionObj --> CS: sessionRecord
deactivate SessionObj
deactivate CS

CS -> HistoryObj: save_message(session_id, "user", user_message)
activate HistoryObj
HistoryObj --> CS: success
deactivate HistoryObj

CS -> HistoryObj: get_messages_by_session(session_id)
activate HistoryObj
HistoryObj --> CS: historyList
deactivate HistoryObj

CS -> CS: _map_role(historyList)

CS -> LLM: invoke(historyList)
activate LLM
LLM --> CS: generatedResponse
deactivate LLM

CS -> HistoryObj: save_message(session_id, "ai", generatedResponse)
activate HistoryObj
HistoryObj --> CS: success
deactivate HistoryObj

opt If this is the first message in the session
    CS -> CS: _generate_title(session_id, user_message)
    CS -> SessionObj: update_title(new_title)
    activate SessionObj
    SessionObj --> CS: success
    deactivate SessionObj
end

CS --> API: responseJSON
deactivate CS

API --> UI: responseJSON
deactivate API

UI -> UI: appendMessageToUI(responseJSON)
UI --> User: Display intelligent chatbot reply
deactivate UI

@enduml
