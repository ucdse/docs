@startuml
title DublinBikes - AI Chatbot Interaction Flow

actor User
participant "chatPage : Chat" as UI
participant "chatRoutes : ChatRoutes" as API
participant "chatService : ChatService" as CS
participant "database : SQLAlchemy ORM" as DB
participant "llmClient : LangChainLLM" as LLM

User -> UI: submitMessage(user_message)
activate UI

UI -> API: chat(session_id, user_message) [Header: Authorization Bearer Token]
activate API

API -> API: _require_auth() (Validates JWT)

API -> CS: generate_chat_response(session_id, user_message, user_id)
activate CS

CS -> CS: _ensure_session(session_id, user_id)
activate CS
CS -> DB: db.session.get() / add()
activate DB
DB --> CS: sessionRecord
deactivate DB
deactivate CS

CS -> LLM: invoke(user_message) via RunnableWithMessageHistory
activate LLM
LLM -> DB: SQLChatMessageHistory: save user msg & load context
activate DB
DB --> LLM: chat context
deactivate DB
LLM --> CS: generatedResponse
deactivate LLM

opt If this is the first message in the session
    CS -> CS: _generate_title(session_id, user_message)
    CS -> DB: db.session.commit() (update Session title)
    activate DB
    DB --> CS: success
    deactivate DB
end

CS --> API: responseJSON
deactivate CS

API --> UI: responseJSON
deactivate API

UI -> UI: setMessages(prev => [...prev, responseJSON])
UI --> User: Display intelligent chatbot reply
deactivate UI

@enduml
