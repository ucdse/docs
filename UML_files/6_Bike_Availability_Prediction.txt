@startuml
title DublinBikes - Bike Availability Prediction Pipeline

actor User
participant "mapsPage : Maps" as UI
participant "stationRoutes : StationRoutes" as API
participant "predictionService : PredictionService" as PS
participant "weatherRecord : WeatherForecast" as WeatherObj
participant "stationRecord : Station" as StationObj
participant "dtModel : DecisionTreeRegressor" as ML


User -> UI: clickStationMarker(station_id)
activate UI

UI -> API: get_station_prediction(station_id)
activate API

API -> PS: get_station_predictions(station_id)
activate PS

PS -> StationObj: query_by_id(station_id)
activate StationObj
StationObj --> PS: station details (capacity, lat, lon)
deactivate StationObj

PS -> WeatherObj: query_future_forecasts(now)
activate WeatherObj
WeatherObj --> PS: List<WeatherForecast>
deactivate WeatherObj

' Construct Features
PS -> PS: format Pandas DataFrame features (time, weather, station)

PS -> ML: predict(df_input)
activate ML
ML --> PS: predictions_array (bikes)
deactivate ML

PS --> API: predictions_list
deactivate PS

API --> UI: predictions_json
deactivate API

' React Declarative State Update
UI -> UI: setPredictionData(predictions_json)

UI --> User: Auto-render Chart Modal
deactivate UI

@enduml
