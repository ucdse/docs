@startuml
title DublinBikes - Bike Availability Prediction Pipeline

actor User
participant "mapsPage : SearchMapsPage" as UI
participant "stationRoutes : StationRoutes" as API
participant "predictionService : PredictionService" as PS
participant "weatherService : WeatherService" as WS
participant "weatherRecord : WeatherForecast" as WeatherObj
participant "availabilityRecord : Availability" as AvailObj
participant "rfModel : RandomForestRegressor" as ML

User -> UI: clickStationMarker(station_id)
activate UI

UI -> API: get_station_prediction(station_id)
activate API

API -> PS: get_station_predictions(station_id)
activate PS

' Needs Weather
PS -> WS: get_weather()
activate WS
WS -> WeatherObj: get_upcoming_forecast()
activate WeatherObj
WeatherObj --> WS: List<WeatherForecast>
deactivate WeatherObj
WS --> PS: weather_context
deactivate WS

' Construct Features
PS -> AvailObj: get_recent_history(station_id)
activate AvailObj
AvailObj --> PS: List<Availability>
deactivate AvailObj

PS -> PS: _construct_features_array(weather_context, availability_history)

PS -> ML: predict(features_array)
activate ML
ML --> PS: predictions_array (bikes, stands)
deactivate ML

PS --> API: predictions_list
deactivate PS

API --> UI: predictions_json
deactivate API

UI -> UI: parsePredictionData(predictions_json)
UI -> UI: renderDataChart()

UI --> User: Display Chart on Map
deactivate UI

@enduml
